  lord_cedric:['Lord Cedric','세드릭 경'],magnus:['Magnus','마그누스'],tormund:['Tormund','토르문드']
};
function koNamedIdentity175(state,name) {
  return Nl(name,state?.namedPatronNames)?.id;
}
function koNamedTemplate175(text,named) {
  if(typeof text!=='string'||!named)return text;
  const aliases=[...new Set([named.name,...(koNamedAliases175[named.id]??[])])];
  const escape=value=>value.replace(/[.*+?^${}()|[\]\\]/g,'\\$&');
  const pattern=new RegExp('(?<![\\p{L}\\p{N}_])(?:'+aliases.map(escape).sort((a,b)=>b.length-a.length).join('|')+')'+'(?:([이가은는을를와과])(?=\\s|의|[.,!?"”’])|(?=에게|한테|의|처럼|[^\\p{L}\\p{N}_]|$))','gu');
  const particles={이:'이(가)',가:'이(가)',은:'은(는)',는:'은(는)',을:'을(를)',를:'을(를)',와:'와(과)',과:'와(과)'};
  return text.replace(pattern,(_,particle)=>'{p}'+(particles[particle]??''));
}
function koNamedLine175(text,named,patron,wife) {
  // One substitution pass: names containing $&, braces, or the other character's
  // canonical name are literal data and cannot become another replacement.
  const template=patron===named?.name?text:koNamedTemplate175(text,named);
  return template.replace(/\{([pw])\}/g,(_,key)=>key==='p'?patron:wife);
}
function koNamedThought175(thought,context) {
  const text=thought.generate(context);
  if(!HP.includes(thought))return text;
  const name=context.recentPatronName,named=Nl(name,context.state?.namedPatronNames);
  return named?koNamedLine175(text,named,name,context.wife.name):text;
}
